为了方便阅读和深入理解，我为您整理了格式化后的完整代码，并添加了**详细的架构级/底层硬件优化注释**（包括 Cache 缓存机制、SIMD 指令分析和内存访问模式）。

### 带有详细注释的完整 C++ 代码

```cpp
/*
 * GEMM 优化阶梯 (GEMM Optimization Ladder)
 * 演示路径: 朴素 IJK -> 循环重排 IKJ -> 分块 Blocked -> AVX-512 FMA 向量化
 *
 * 白板核心讨论点 (Whiteboard Talking Points):
 * 1. 行主序 (Row-major) 下，B[k][j] 在 I-J-K 朴素循环中，最内层 k 循环每次步长为 N，
 *    导致极其严重的 Cache Miss (缓存未命中)。
 * 2. IKJ 循环重排：将 A[i][k] 提至寄存器；最内层 j 循环连续顺序访问 B[k][] 和 C[i][]，
 *    触发 CPU 的空间局部性 (Spatial Locality) 与硬件预取 (Hardware Prefetch)。
 * 3. 分块 (Blocked) GEMM：限制子矩阵大小，使数据能完全驻留在 L1/L2 Cache 中，实现时间局部性 (Temporal Reuse)。
 * 4. AVX-512 FMA：单指令同时处理 16 个单精度浮点数 (float) 的乘加运算 (Fused Multiply-Add)，极大提升 GFLOPS。
 */

#include <iostream>
#include <vector>
#include <chrono>
#include <cmath>
#include <algorithm>
#include <random>
#include <cstring>
#include <cassert>

// 如果定义了 GEMM_AVX512 宏，引入 Intel AVX-512 Intrinsics 头文件
#if defined(GEMM_AVX512)
#include <immintrin.h>
#endif

// 使用 steady_clock 进行高精度性能计时
using Clock = std::chrono::steady_clock;

/**
 * @brief 计算从 start 到当前时间的毫秒数
 */
static double elapsed_ms(Clock::time_point start) {
    return std::chrono::duration<double, std::milli>(Clock::now() - start).count();
}

/**
 * 1. 朴素 IJK 顺序矩阵乘法 (Naive IJK)
 *
 * 【性能瓶颈分析】：
 * C[i][j] = sum(A[i][k] * B[k][j])
 * 最内层循环是 k：
 * - A[i][k] 的访问是连续的 (A[i*n + k]) -> 好
 * - B[k][j] 的访问跨度为 n (B[k*n + j]) -> 极其糟糕！
 *   在行主序存储下，跨行访问导致每次 Iteration 都可能触发 L1/L2 Cache Miss，
 *   内存带宽成为严重瓶颈。
 */
static void gemm_naive_ijk(const std::vector<float>& A, const std::vector<float>& B, std::vector<float>& C, int n) {
    std::fill(C.begin(), C.end(), 0.0f);
    for (int i = 0; i < n; ++i) {
        for (int j = 0; j < n; ++j) {
            float sum = 0.0f;
            for (int k = 0; k < n; ++k) {
                // B[k * n + j] 每次 k+1，地址跳跃 n * sizeof(float) 字节
                sum += A[static_cast<size_t>(i) * n + k] * B[static_cast<size_t>(k) * n + j];
            }
            C[static_cast<size_t>(i) * n + j] = sum;
        }
    }
}

/**
 * 2. IKJ 循环重排矩阵乘法 (IKJ Reordered)
 *
 * 【优化原理】：
 * 将 k 循环提升到中层，j 循环放到最内层：
 * - 标量 A[i][k] 在最内层 j 循环中是常数，可直接被编译器提至 CPU 寄存器 (Register Hoisting)。
 * - 最内层 j 循环中：
 *   - B[k*n + j] 是连续内存访问（步长为 1）。
 *   - C[i*n + j] 也是连续内存访问（步长为 1）。
 * - 极大地提升了 Cache 命中率并能触发硬件数据预取 (Hardware Prefetching)。
 */
static void gemm_ikj(const std::vector<float>& A, const std::vector<float>& B, std::vector<float>& C, int n) {
    std::fill(C.begin(), C.end(), 0.0f);
    for (int i = 0; i < n; ++i) {
        for (int k = 0; k < n; ++k) {
            // A[i][k] 在内层 j 循环中保持不变，广播/寄存器复用
            const float r = A[static_cast<size_t>(i) * n + k];
            for (int j = 0; j < n; ++j) {
                // C 和 B 均为连续地址累加，对 L1 Cache 极其友好
                C[static_cast<size_t>(i) * n + j] += r * B[static_cast<size_t>(k) * n + j];
            }
        }
    }
}

/**
 * 3. 标量分块矩阵乘法 (Blocked / Tiled Scalar GEMM)
 *
 * 【优化原理】：
 * 当矩阵维度 N 非常大（超出 L1/L2 Cache 容量）时，先前的 Cache 会被反复刷掉 (Cache Thrashing)。
 * 通过引入三重分块循环 (si, sk, sj)，将大矩阵切分为 block_size x block_size 的小 Tile：
 * - 确保当前的 A/B/C 子块能完全放入 L1/L2 Cache 中。
 * - 提高时间局部性 (Temporal Locality)，子块在 Cache 中被充分反复利用后才被换出。
 */
static void gemm_blocked_scalar(const std::vector<float>& A, const std::vector<float>& B, std::vector<float>& C, int n, int block_size) {
    std::fill(C.begin(), C.end(), 0.0f);
    // 外层 3 重循环：控制 Tile / Block 的起始偏移
    for (int sj = 0; sj < n; sj += block_size) {
        for (int sk = 0; sk < n; sk += block_size) {
            for (int si = 0; si < n; si += block_size) {
                // 计算当前 Block 的边界，防止边界溢出
                const int i_end = std::min(si + block_size, n);
                const int k_end = std::min(sk + block_size, n);
                const int j_end = std::min(sj + block_size, n);

                // 内层 3 重 IKJ 循环：在 L1/L2 Cache 内部进行高效率计算
                for (int i = si; i < i_end; ++i) {
                    for (int k = sk; k < k_end; ++k) {
                        const float r = A[static_cast<size_t>(i) * n + k];
                        for (int j = sj; j < j_end; ++j) {
                            C[static_cast<size_t>(i) * n + j] += r * B[static_cast<size_t>(k) * n + j];
                        }
                    }
                }
            }
        }
    }
}

#if defined(GEMM_AVX512)
/**
 * 4. AVX-512 向量化 + 分块矩阵乘法 (AVX-512 SIMD Blocked GEMM)
 *
 * 【硬件级优化】：
 * - _mm512_set1_ps: 将标量 A[i][k] 广播填充到 512-bit SIMD 寄存器的 16 个 float 通道。
 * - _mm512_loadu_ps: 一次性从内存中未对齐加载 16 个连续的 float (512 bit)。
 * - _mm512_fmadd_ps: 乘加融合指令 (FMA)，单周期内完成 vec_c = (vec_a * vec_b) + vec_c。
 * - 尾块处理 (Remainder Loop)：处理 j 维度不能被 16 整除的剩余部分。
 */
static void gemm_blocked_avx512(const std::vector<float>& A, const std::vector<float>& B, std::vector<float>& C, int n, int block_size) {
    std::fill(C.begin(), C.end(), 0.0f);
    for (int sj = 0; sj < n; sj += block_size) {
        for (int sk = 0; sk < n; sk += block_size) {
            for (int si = 0; si < n; si += block_size) {
                const int i_end = std::min(si + block_size, n);
                const int k_end = std::min(sk + block_size, n);
                const int j_end = std::min(sj + block_size, n);

                for (int i = si; i < i_end; ++i) {
                    for (int k = sk; k < k_end; ++k) {
                        // 将 A[i][k] 广播到 512 位寄存器的全部 16 个 float 通道中
                        const __m512 vec_a = _mm512_set1_ps(A[static_cast<size_t>(i) * n + k]);
                        
                        int j = sj;
                        // 向量化主循环：每次处理 16 个 float (512 bits)
                        for (; j + 16 <= j_end; j += 16) {
                            float* c_ptr = &C[static_cast<size_t>(i) * n + j];
                            float* b_ptr = &B[static_cast<size_t>(k) * n + j];

                            __m512 vec_c = _mm512_loadu_ps(c_ptr); // 加载 C 的 16 个元素
                            __m512 vec_b = _mm512_loadu_ps(b_ptr); // 加载 B 的 16 个元素

                            // FMA 指令: vec_c = vec_a * vec_b + vec_c
                            vec_c = _mm512_fmadd_ps(vec_a, vec_b, vec_c);

                            _mm512_storeu_ps(c_ptr, vec_c); // 写回 C
                        }

                        // 标量尾块循环 (Scalar Cleanup Loop): 处理剩余不足 16 个的元素
                        for (; j < j_end; ++j) {
                            C[static_cast<size_t>(i) * n + j] += A[static_cast<size_t>(i) * n + k] * B[static_cast<size_t>(k) * n + j];
                        }
                    }
                }
            }
        }
    }
}
#endif

/**
 * @brief 计算两个矩阵的最大绝对误差，用于数值正确性校验
 */
static float max_abs_diff(const std::vector<float>& a, const std::vector<float>& b) {
    float m = 0.0f;
    for (size_t i = 0; i < a.size(); ++i) {
        m = std::max(m, std::abs(a[i] - b[i]));
    }
    return m;
}

/**
 * @brief 生成范围在 [-1.0, 1.0] 之间的随机 N x N 矩阵
 */
static std::vector<float> random_matrix(int n, std::mt19937& rng) {
    std::uniform_real_distribution<float> dist(-1.0f, 1.0f);
    std::vector<float> m(static_cast<size_t>(n) * n);
    for (float& v : m) {
        v = dist(rng);
    }
    return m;
}

/**
 * @brief 执行测试用例，运行各种 GEMM 实现并对比性能与精度
 */
static void run_case(const std::vector<float>& A, const std::vector<float>& B, int n) {
    // 使用准确且高效的 ikj 作为基准 (Reference) 计算正确答案
    std::vector<float> ref(static_cast<size_t>(n) * n, 0.0f);
    gemm_ikj(A, B, ref, n);

    // 通用基准测试 Lambda 表达式
    auto bench = [&](auto fn, const char* name) {
        std::vector<float> C(static_cast<size_t>(n) * n);
        const auto t0 = Clock::now();
        fn(A, B, C, n);
        const double ms = elapsed_ms(t0);
        const float diff = max_abs_diff(C, ref);

        // 格式化输出对齐
        std::cout << "  " << name;
        for (int pad = static_cast<int>(strlen(name)); pad < 18; ++pad) {
            std::cout << ' ';
        }
        std::cout << ms << " ms, max diff " << diff << "\n";
        
        // 确保计算结果精度满足要求
        assert(diff < 1e-3f);
    };

    std::cout << "N=" << n << ":\n";

    // 1. 测试朴素算法
    bench([&](const std::vector<float>& a, const std::vector<float>& b, std::vector<float>& c, int size) {
        gemm_naive_ijk(a, b, c, size);
    }, "naive_ijk");

    // 2. 测试 IKJ 重排算法
    bench([&](const std::vector<float>& a, const std::vector<float>& b, std::vector<float>& c, int size) {
        gemm_ikj(a, b, c, size);
    }, "ikj");

    // 3. 测试标量分块算法 (Block Size = 64)
    bench([&](const std::vector<float>& a, const std::vector<float>& b, std::vector<float>& c, int size) {
        gemm_blocked_scalar(a, b, c, size, 64);
    }, "blocked(64)");

#if defined(GEMM_AVX512)
    // 4. 测试 AVX-512 分块向量化算法
    bench([&](const std::vector<float>& a, const std::vector<float>& b, std::vector<float>& c, int size) {
        gemm_blocked_avx512(a, b, c, size, 64);
    }, "blocked_avx512");
#endif
}

int main() {
    std::mt19937 rng(0);
    const int N = 256;
    const std::vector<float> A = random_matrix(N, rng);
    const std::vector<float> B = random_matrix(N, rng);

    std::cout << "=== GEMM optimization ladder ===\n";

#if defined(GEMM_AVX512)
    std::cout << "AVX-512 FMA enabled\n";
#else
    std::cout << "Scalar only (build with -DGEMM_AVX512 -mavx512f for SIMD path)\n";
#endif

    run_case(A, B, N);

    std::cout << "gemm: all tests passed\n";
    return 0;
}
```

### 核心优化效果小结

1. **`naive_ijk` $\rightarrow$ `ikj**`：仅通过**改变循环顺序**，将内存访问改为连续的步长 1（Stride 1），通常能带来 **5x - 10x** 的巨大性能飞跃。
2. **`ikj` $\rightarrow$ `blocked**`：通过 **Tiling（分块）** 控制数据驻留在 L1/L2 Cache，当矩阵尺寸 $N \ge 1024$ 时性能提升非常显著。
3. **`blocked` $\rightarrow$ `blocked_avx512**`：利用 **SIMD 寄存器与 FMA 乘加指令**，单周期算力吞吐放大 **16 倍**。
